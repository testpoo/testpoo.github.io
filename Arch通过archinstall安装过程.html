2.archinstall安装

# 根据提示安装系统
archinstall

# 如果执行速度慢，可以换源
reflector -c China -a 6 --sort rate --save /etc/pacman.d/mirrorlist

3.添加Arch Linux CN 软件仓库源

# 在/etc/pacman.conf文件最后面添加
[archlinuxcn]
Server = https://mirrors.tuna.tsinghua.edu.cn/archlinuxcn/$arch

# 然后安装archlinuxcn-keyring
pacman -Sy archlinuxcn-keyring

6.2. 输入法环境变量设置

启用fcitx输入需要配置环境变量：

nano /etc/environment

XIM="fcitx"
#GTK_IM_MODULE=fcitx
QT_IM_MODULE=fcitx
XMODIFIERS="@im=fcitx"
INPUT_METHOD=fcitx
SDL_IM_MODULE=fcitx

6.3. 启动Sway

# 编辑 ~/.bash_profile加入以下内容即可：

[ "$(tty)" = "/dev/tty1" ] && exec sway

# 编辑~/.config/sway

6.5. 设置终端(foot)字体大小

cp -r /etc/xdg/foot/foot.ini ~/.config/foot/

font=FreeMono:size=12

WeChat: xcb-util-image

sudo pacman -Rns alacritty htop smartmontools vim wget lightdm lightdm-gtk-greeter

7.2. 启动Labwc

# 编辑 ~/.bash_profile加入以下内容即可：

[ "$(tty)" = "/dev/tty1" ] && exec labwc

7.3.设置 Fcitx5 初始配置

配置 Group 直接启动 fcitx5 是只有西文键盘的，把下面的内容粘贴到 ~/.config/fcitx5/profile

[Groups/0]
# Group Name
Name=Default
# Layout
Default Layout=us
# Default Input Method
DefaultIM=rime

[Groups/0/Items/0]
# Name
Name=rime
# Layout
Layout=

[GroupOrder]
0=Default

DefaultIM=xx 为设置默认输入法，后面的Group中的Name为具体的输入法名字，按0，1，2……这样的编号排序，修改文件时，在要fcitx5关闭状态下，否则修改不生效。

7.4. iwd无法联网

# 当前不需要设置了，可以直接使用

sudo systemctl enable systemd-resolved.service --now
systemctl restart iwd

9.pacman命令

pacman 是 Arch Linux 的包管理器。

9.1.系统更新

# 更新所有已安装的软件包
sudo pacman -Syu

# 仅更新软件包数据库(不实际更新软件)
sudo pacman -Sy

9.2.软件包安装

# 安装单个软件包
sudo pacman -S 包名

# 安装多个软件包
sudo pacman -S 包名1 包名2 包名3

# 安装本地软件包文件(不处理依赖)
sudo pacman -U /路径/到/包名.pkg.tar.zst

9.3.软件包删除

# 删除软件包但保留其依赖
sudo pacman -R 包名

# 删除软件包及其不再需要的依赖
sudo pacman -Rs 包名

# 删除软件包及其所有配置文件
sudo pacman -Rns 包名

9.4.软件包查询

# 搜索软件包(名称和描述)
pacman -Ss 关键词

# 列出已安装的软件包
pacman -Q

# 查询某个文件属于哪个软件包
pacman -Qo /路径/到/文件

# 显示软件包详细信息
pacman -Qi 包名

9.5.清理缓存

# 清理未安装软件包的缓存
sudo pacman -Sc

# 清理所有软件包的缓存(包括已安装的)
sudo pacman -Scc
